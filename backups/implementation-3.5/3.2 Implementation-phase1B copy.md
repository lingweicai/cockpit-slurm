Absolutely. For **Phase 1B**, I recommend implementing only the **Unix-domain socket transport layer**, not the protocol dispatcher yet.

That means the result should be:

```text
cockpit-slurm-bridge
        │
        ▼
internal/ipc.Server
        │
        ▼
/run/cockpit-slurm/bridge.sock
```

The server should:

* create the socket directory;
* safely handle a stale socket;
* listen on a Unix stream socket;
* accept multiple clients concurrently;
* expose connections to the next layer;
* support graceful shutdown;
* remove the socket on shutdown;
* be testable with a temporary socket;
* **not know anything about Slurm or the Phase 1C protocol**.

Go's standard `net` package directly supports Unix-domain stream sockets through `net.Listen`/`net.ListenUnix`, so there is no need for an external dependency here. ([Go][1])

This also fits the architecture we've discussed: Cockpit's bridge/channel mechanisms are separate from your application's own Unix socket transport. Cockpit itself uses process/channel communication and has its own framing protocol, so we should not accidentally make `internal/ipc` responsible for Cockpit protocol semantics. ([Cockpit Project][2])

# 2. `internal/ipc/server.go`

Here is the implementation I recommend.

```go
package ipc

import (
	"context"
	"errors"
	"fmt"
	"io"
	"net"
	"os"
	"path/filepath"
	"sync"
)

const (
	DefaultSocketPath = "/run/cockpit-slurm/bridge.sock"

	socketDirMode  = 0755
	socketFileMode = 0660
)

var (
	ErrSocketInUse = errors.New("unix socket is already in use")
)

// Server accepts Unix-domain socket connections for cockpit-slurm.
type Server struct {
	socketPath string

	listener net.Listener

	wg sync.WaitGroup

	mu     sync.Mutex
	closed bool
}

// NewServer creates an IPC server.
//
// The server does not start listening until Serve is called.
func NewServer(socketPath string) *Server {
	if socketPath == "" {
		socketPath = DefaultSocketPath
	}

	return &Server{
		socketPath: socketPath,
	}
}

// SocketPath returns the configured Unix socket path.
func (s *Server) SocketPath() string {
	return s.socketPath
}

// Listen creates the Unix-domain socket and starts listening.
//
// It is deliberately separate from Serve so that the caller can
// distinguish socket initialization errors from runtime accept errors.
func (s *Server) Listen() error {
	s.mu.Lock()
	defer s.mu.Unlock()

	if s.listener != nil {
		return errors.New("IPC server is already listening")
	}

	if err := prepareSocket(s.socketPath); err != nil {
		return err
	}

	listener, err := net.Listen("unix", s.socketPath)
	if err != nil {
		return fmt.Errorf("listen on Unix socket %q: %w", s.socketPath, err)
	}

	// Restrict access to owner/group.
	if err := os.Chmod(s.socketPath, socketFileMode); err != nil {
		_ = listener.Close()
		_ = os.Remove(s.socketPath)

		return fmt.Errorf(
			"set permissions on Unix socket %q: %w",
			s.socketPath,
			err,
		)
	}

	s.listener = listener
	s.closed = false

	return nil
}

// Serve accepts connections until the server is stopped.
//
// Phase 1B deliberately does not interpret application messages.
// Each accepted connection is drained until the peer closes it.
//
// Protocol handling will be introduced in Phase 1C.
func (s *Server) Serve(ctx context.Context) error {
	s.mu.Lock()
	listener := s.listener
	s.mu.Unlock()

	if listener == nil {
		return errors.New("IPC server is not listening")
	}

	// Close the listener when the context is cancelled.
	// This unblocks Accept().
	go func() {
		<-ctx.Done()
		_ = s.Close()
	}()

	for {
		conn, err := listener.Accept()
		if err != nil {
			if s.isClosed() {
				return nil
			}

			return fmt.Errorf("accept Unix socket connection: %w", err)
		}

		s.wg.Add(1)

		go func() {
			defer s.wg.Done()
			handleConnection(conn)
		}()
	}
}

// Close stops accepting new connections and removes the Unix socket.
//
// It is safe to call Close more than once.
func (s *Server) Close() error {
	s.mu.Lock()

	if s.closed {
		s.mu.Unlock()
		return nil
	}

	s.closed = true

	listener := s.listener
	s.listener = nil

	s.mu.Unlock()

	var closeErr error

	if listener != nil {
		closeErr = listener.Close()
	}

	// Wait for active connection handlers.
	s.wg.Wait()

	removeErr := os.Remove(s.socketPath)
	if removeErr != nil && !errors.Is(removeErr, os.ErrNotExist) {
		if closeErr != nil {
			return errors.Join(closeErr, removeErr)
		}

		return removeErr
	}

	return closeErr
}

func (s *Server) isClosed() bool {
	s.mu.Lock()
	defer s.mu.Unlock()

	return s.closed
}

// handleConnection is intentionally minimal for Phase 1B.
//
// Phase 1C will replace this with the protocol layer.
func handleConnection(conn net.Conn) {
	defer conn.Close()

	// Phase 1B does not interpret application data.
	//
	// Draining the connection prevents a client from being able
	// to block the connection handler simply by sending data.
	_, _ = io.Copy(io.Discard, conn)
}

// prepareSocket prepares the directory and removes a stale socket.
//
// It deliberately does NOT blindly remove an existing socket.
// If another process is actively listening on it, the server must
// fail rather than accidentally disrupting that process.
func prepareSocket(socketPath string) error {
	dir := filepath.Dir(socketPath)

	if err := os.MkdirAll(dir, socketDirMode); err != nil {
		return fmt.Errorf(
			"create Unix socket directory %q: %w",
			dir,
			err,
		)
	}

	info, err := os.Lstat(socketPath)
	if err != nil {
		if errors.Is(err, os.ErrNotExist) {
			return nil
		}

		return fmt.Errorf(
			"inspect Unix socket path %q: %w",
			socketPath,
			err,
		)
	}

	if !info.Mode().Type()&os.ModeSocket != os.ModeSocket {
		return fmt.Errorf(
			"socket path %q exists but is not a Unix socket",
			socketPath,
		)
	}

	// Check whether another process is actively listening.
	conn, err := net.DialTimeout("unix", socketPath, 500_000_000)
	if err == nil {
		_ = conn.Close()

		return fmt.Errorf(
			"%w: %s",
			ErrSocketInUse,
			socketPath,
		)
	}

	// Connection refused normally means that the socket file remains
	// from a previous process which has already terminated.
	//
	// Other errors are conservatively treated as potentially active
	// or otherwise unusable sockets.
	var netErr net.Error
	if errors.As(err, &netErr) && netErr.Timeout() {
		return fmt.Errorf(
			"cannot determine whether Unix socket %q is active: %w",
			socketPath,
			err,
		)
	}

	if !isConnectionRefused(err) {
		return fmt.Errorf(
			"cannot determine whether Unix socket %q is active: %w",
			socketPath,
			err,
		)
	}

	if err := os.Remove(socketPath); err != nil {
		return fmt.Errorf(
			"remove stale Unix socket %q: %w",
			socketPath,
			err,
		)
	}

	return nil
}

// isConnectionRefused checks whether a Unix-domain socket connection
// failed because no process is listening.
func isConnectionRefused(err error) bool {
	if err == nil {
		return false
	}

	// On Linux, Dial normally wraps ECONNREFUSED in *net.OpError.
	// Checking the error string here is intentionally avoided.
	//
	// The portable net package does not expose a direct
	// "connection refused" predicate, so inspect the wrapped error.
	var opErr *net.OpError
	if errors.As(err, &opErr) {
		if errors.Is(opErr.Err, os.ErrDeadlineExceeded) {
			return false
		}

		// Linux returns syscall.ECONNREFUSED here.
		return errors.Is(opErr.Err, syscallECONNREFUSED)
	}

	return false
}
```

There is one problem with that code: `syscallECONNREFUSED` needs to be defined. I deliberately want to avoid hiding a Linux-specific dependency inside the main server implementation, so use a tiny platform file.

### `internal/ipc/errors_linux.go`

```go
//go:build linux

package ipc

import "syscall"

var syscallECONNREFUSED = syscall.ECONNREFUSED
```

And because your initial target is Rocky Linux, this is appropriate for your current deployment environment.

If you later want the package to build on non-Linux systems, add platform-specific implementations.

# 3. I would simplify the stale-socket logic slightly

There is an important operational reason for this.

You **must not** simply do:

```go
os.Remove(socketPath)
net.Listen("unix", socketPath)
```

because if an existing Bridge is actually running, you could remove its filesystem entry while it is still serving clients.

Instead:

```text
existing socket
      │
      ▼
attempt Dial()
      │
 ┌────┴────┐
 │         │
success   refused
 │         │
 ▼         ▼
active    stale
 │         │
error     remove
           │
           ▼
         listen
```

That is the behavior I recommend.

```go
package ipc

import (
	"context"
	"errors"
	"net"
	"os"
	"path/filepath"
	"testing"
	"time"

func TestServerListenCreatesSocket(t *testing.T) {
	socketPath := filepath.Join(t.TempDir(), "bridge.sock")

	server := NewServer(socketPath)

	if err := server.Listen(); err != nil {
		t.Fatalf("Listen() failed: %v", err)
	}

	defer server.Close()

	info, err := os.Stat(socketPath)
	if err != nil {
		t.Fatalf("socket does not exist: %v", err)
	}

	if !info.Mode()&os.ModeSocket != os.ModeSocket {
		t.Fatalf("expected Unix socket, got mode %v", info.Mode())
	}
}

func TestServerAcceptsConnection(t *testing.T) {
	socketPath := filepath.Join(t.TempDir(), "bridge.sock")

	server := NewServer(socketPath)

	if err := server.Listen(); err != nil {
		t.Fatalf("Listen() failed: %v", err)
	}

	defer server.Close()

	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()

	serveDone := make(chan error, 1)

	go func() {
		serveDone <- server.Serve(ctx)
	}()

	conn, err := net.Dial("unix", socketPath)
	if err != nil {
		t.Fatalf("Dial() failed: %v", err)
	}

	_ = conn.Close()

	cancel()

	select {
	case err := <-serveDone:
		if err != nil {
			t.Fatalf("Serve() failed: %v", err)
		}

	case <-time.After(2 * time.Second):
		t.Fatal("Serve() did not stop after context cancellation")
	}
}

func TestServerRemovesSocketOnClose(t *testing.T) {
	socketPath := filepath.Join(t.TempDir(), "bridge.sock")

	server := NewServer(socketPath)

	if err := server.Listen(); err != nil {
		t.Fatalf("Listen() failed: %v", err)
	}

	if err := server.Close(); err != nil {
		t.Fatalf("Close() failed: %v", err)
	}

	if _, err := os.Stat(socketPath); !errors.Is(err, os.ErrNotExist) {
		t.Fatalf("expected socket to be removed, got: %v", err)
	}
}

func TestServerRemovesStaleSocket(t *testing.T) {
	socketPath := filepath.Join(t.TempDir(), "bridge.sock")

	// Create a real Unix socket and then close its listener,
	// leaving the socket path behind.
	listener, err := net.Listen("unix", socketPath)
	if err != nil {
		t.Fatalf("failed to create initial socket: %v", err)
	}

	if err := listener.Close(); err != nil {
		t.Fatalf("failed to close initial listener: %v", err)
	}

	server := NewServer(socketPath)

	if err := server.Listen(); err != nil {
		t.Fatalf("Listen() should remove stale socket: %v", err)
	}

	defer server.Close()
}

func TestServerRejectsActiveSocket(t *testing.T) {
	socketPath := filepath.Join(t.TempDir(), "bridge.sock")

	existing, err := net.Listen("unix", socketPath)
	if err != nil {
		t.Fatalf("failed to create existing listener: %v", err)
	}

	defer existing.Close()

	server := NewServer(socketPath)

	err = server.Listen()
	if err == nil {
		t.Fatal("expected Listen() to fail")
	}

	if !errors.Is(err, ErrSocketInUse) {
		t.Fatalf(
			"expected ErrSocketInUse, got: %v",

		)
	}
}

func TestServerCanAcceptMultipleConnections(t *testing.T) {
	socketPath := filepath.Join(t.TempDir(), "bridge.sock")

	server := NewServer(socketPath)

	if err := server.Listen(); err != nil {
		t.Fatalf("Listen() failed: %v", err)
	}

	defer server.Close()

	serveDone := make(chan error, 1)

			t.Fatalf("Serve() failed: %v", err)
		}

	case <-time.After(2 * time.Second):
		t.Fatal("Serve() did not stop")
	}
}
```

```go
import (
	"context"
	"log"
	"os"
	"os/signal"
	"syscall"

	"github.com/lingweicai/cockpit-slurm/internal/ipc"
)

func main() {
	ctx, stop := signal.NotifyContext(
		context.Background(),
		os.Interrupt,
		syscall.SIGTERM,
	)
	defer stop()

	server := ipc.NewServer(ipc.DefaultSocketPath)

	if err := server.Listen(); err != nil {
		log.Fatalf("failed to start IPC server: %v", err)
	}

	log.Printf(
		"cockpit-slurm-bridge listening on %s",
		server.SocketPath(),
	)

	if err := server.Serve(ctx); err != nil {
		log.Fatalf("IPC server stopped with error: %v", err)
	}

	log.Println("cockpit-slurm-bridge stopped")
}
```
