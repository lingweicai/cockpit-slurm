## Socket

### socket path

For the socket path between IPC server and cockpit.channel(), we would have two installation options:

* Option A, the cockpit-slurm to be installed by root, socket proposed: /run/cockpit-slurm/cockpit-slurm.sock  
* Option B, the cockpit-slurm to be installed by a development user, socket path proposed: /home/user/.local/run/cockpit-slurm.sock 

We need to substitute user with the real user login name running the application in Option B.

### socket permissions

Because cockpit-bridge acts as the direct user proxy connecting to your systemd socket, you must ensure two things for a standard stream connection to work:
* Socket Permissions: By default, systemd might create sockets owned by root:root with restrictive permissions (0600). If standard, non-root users log into your Cockpit dashboard, they will get a "Permission Denied" error. Fix this in your systemd .socket unit file:ini

```ini
[Socket]
ListenStream=/run/cockpit-slurm/cockpit-slurm.sock
SocketMode=0660
SocketGroup=cockpit-slurm # Or whichever group your Cockpit users share
```
## System Daemon Path
* If the cockpit-slurm was installed by root, the unit file is like: 

```ini
[Unit]
Description=Cockpit Slurm Persistent Service
After=slurmctld.service
Wants=slurmctld.service

[Service]
Type=simple
ExecStart=/usr/local/bin/cockpit-slurm
Restart=on-failure
RestartSec=5

User=cockpit-slurm
Group=cockpit-slurm

RuntimeDirectory=cockpit-slurm
RuntimeDirectoryMode=0750

[Install]
WantedBy=multi-user.target
```
* if the daemon to be installed and start by a user, the execute path would be ExecStart=/home/usr/.local/bin/cockpit-slurm.  User can run the executable directly on a terminal as well in addition of systemd daemon.  

## Implementation for backend IPC Server

```go
package main

import (
	"bufio"
	"fmt"
	"log"
	"net"
	"os"
	"os/user"
	"strings"
	"os/signal"
	"syscall"
)

const DefaultSocketPath = "/run/cockpit-slurm/cockpit-slurm.sock"
const UserSocketPathPattern = "/run/user/%s/cockpit-slurm/cockpit-slurm.sock"

// getSocketPath determines the socket path based on user identity and groups.
func getSocketPath() (string, error) {
	// 1. Get the current logged-in user
	currentUser, err := user.Current()
	if err != nil {
		return "", fmt.Errorf("failed to get current user: %w", err)
	}

	// 2. Get all group IDs the user belongs to
	groupIds, err := currentUser.GroupIds()
	if err != nil {
		return "", fmt.Errorf("failed to get user groups: %w", err)
	}

	// 3. Check if any of those group IDs resolve to the name "wheel"
	isWheelMember := false
	for _, gid := range groupIds {
		g, err := user.LookupGroupId(gid)
		if err == nil && g.Name == "wheel" {
			isWheelMember = true
			break
		}
	}

	// 4. Return the appropriate path
	if isWheelMember {
		return DefaultSocketPath, nil
	}

	// Replace the placeholder pattern with the active user's UID
	return fmt.Sprintf(UserSocketPathPattern, currentUser.Uid), nil
}

func main() {
  socketPath, err := getSocketPath()
  if err != nil {
		log.Fatalf("Initialization error: %v", err)
	}
  // 1. Clean up the socket file if it already exists from a previous crash
if err := os.RemoveAll(socketPath); err != nil {
		log.Fatalf("Failed to clean up old socket: %v", err)
	}

	// 2. Bind to the Unix domain socket
	listener, err := net.Listen("unix", socketPath)
	if err != nil {
		log.Fatalf("Failed to listen on socket %s: %v", socketPath, err)
	}
	defer listener.Close()

	// 3. Adjust permissions so Cockpit users/groups can write to it
	// (Alternatively, handle this via systemd's SocketMode configuration)
	if err := os.Chmod(socketPath, 0660); err != nil {
		log.Fatalf("Failed to set socket permissions: %v", err)
	}

	log.Printf("Backend daemon started. Listening on %s...", socketPath)

	// 4. Clean up the socket file gracefully upon system exit/signals
	setupSignalHandler(listener)

	// 5. Connection loop: accept incoming Cockpit tabs concurrently
	for {
		conn, err := listener.Accept()
		if err != nil {
			log.Printf("Error accepting connection: %v", err)
			continue
		}

		// Spin up a new goroutine for every single Cockpit tab connection
		go handleCockpitTab(conn)
	}
}

// Handles data streams isolated to a specific Cockpit tab
func handleCockpitTab(conn net.Conn) {
	defer conn.Close()
	
	// Create a unique identifier for logging purposes
	remoteAddr := conn.RemoteAddr().String()
	log.Printf("[Tab %s] Connected", remoteAddr)

	// Use a scanner to read incoming messages line-by-line (\n)
	scanner := bufio.NewScanner(conn)
	for scanner.Scan() {
		message := scanner.Text()
		log.Printf("[Tab %s] Received: %s", remoteAddr, message)

		// Example Application Logic: Echo or command processing
		switch message {
		case "PING":
			// Write back down the raw stream. Make sure to include \n 
			// so the frontend can easily slice message frames.
			fmt.Fprintln(conn, "PONG")
		case "GET_STATUS":
			fmt.Fprintln(conn, "STATUS: RUNNING")
		default:
			fmt.Fprintf(conn, "UNKNOWN_COMMAND: %s\n", message)
		}
	}

	if err := scanner.Err(); err != nil {
		log.Printf("[Tab %s] Error reading stream: %v", remoteAddr, err)
	}

	log.Printf("[Tab %s] Disconnected / Closed tab", remoteAddr)
}

func setupSignalHandler(listener net.Listener) {
	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		<-sigChan
		log.Println("Shutting down daemon...")
		listener.Close()
		os.Remove(socketPath)
		os.Exit(0)
	}()
}
```

```TS
import cockpit from "cockpit";

const DEFAULT_SOCKET_PATH = "/run/cockpit-slurm/cockpit-slurm.sock";
const USER_SOCKET_PATH_PATTERN = (uid: number | string) => `/run/user/${uid}/cockpit-slurm/cockpit-slurm.sock`;

interface ChannelOptions {
    payload: string;
    unix: string;
    binary: string;
}

/**
 * Initializes Cockpit and sets up the communication channel, 
 * falling back to the user-specific socket if the global one fails.
 */
async function initializeSlurmChannel(): Promise<any> {
    // 1. Always await initialization before accessing cockpit.info
    await cockpit.init();

    // 2. Extract the current user's UNIX UID
    const userUid = cockpit.info?.user?.uid;
    if (!userUid) {
        throw new Error("Unable to retrieve user UID from cockpit.info");
    }

    console.log(`System user identified with UID: ${userUid}`);

    return new Promise((resolve, reject) => {
        console.log(`Attempting connection to system socket: ${DEFAULT_SOCKET_PATH}`);
        
        // 3. Try connecting to the primary system-wide socket
        const mainChannel = cockpit.channel({
            payload: "stream",
            unix: DEFAULT_SOCKET_PATH,
            binary: "raw",
        } as ChannelOptions);

        // Track if this primary channel has encountered an error
        let primaryFailed = false;

        mainChannel.onready = () => {
            if (!primaryFailed) {
                console.log("Successfully connected to the system-wide socket.");
                resolve(mainChannel);
            }
        };

        mainChannel.onclose = (options: { problem?: string }) => {
            // If the connection drops or fails before opening, switch to fallback
            if (options?.problem) {
                primaryFailed = true;
                console.warn(`System socket inaccessible (${options.problem}). Switching to user fallback...`);
                
                // Close the broken channel resource completely
                mainChannel.close();

                // 4. Construct the fallback user-specific path
                const fallbackPath = USER_SOCKET_PATH_PATTERN(userUid);
                console.log(`Attempting connection to user fallback socket: ${fallbackPath}`);

                const fallbackChannel = cockpit.channel({
                    payload: "stream",
                    unix: fallbackPath,
                    binary: "raw",
                } as ChannelOptions);

                fallbackChannel.onready = () => {
                    console.log("Successfully connected to the user-specific socket.");
                    resolve(fallbackChannel);
                };

                fallbackChannel.onclose = (fallbackOptions: { problem?: string }) => {
                    reject(new Error(`Both sockets failed. Fallback error: ${fallbackOptions?.problem || "unknown"}`));
                };
            }
        };
    });
}

// Usage Example:
async function startApp() {
    try {
        const activeChannel = await initializeSlurmChannel();
        
        // You can now safely use your channel!
        activeChannel.onmessage = (data: any) => {
            console.log("Received data from backend:", data);
        };
        
        activeChannel.send("Hello from Cockpit Frontend!");
    } catch (error) {
        console.error("Failed to establish any backend socket connection:", error);
    }
}

startApp();
```
